628sub pvalue ($array1, $array2) {
629 my $current_sub = (split(/::/,(caller(0))[3]))[-1]; # https://stackoverflow.com/questions/2559792/how-can-i-get-the-name-of-the-current-subroutine-in-perl
630 if (grep {!defined} @$array1) {
631 p $array1, array_max => scalar @{ $array1 };
632 die "undefined values found in $current_sub"
633 }
634 if (grep {!defined} @$array2) {
635 p $array2;
636 die "undefined values found in $current_sub";
637 }
638 return 1.0 if scalar @$array1 <= 1;
639 return 1.0 if scalar @$array2 <= 1;
640 my $mean1 = sum(@{ $array1 });
641 $mean1 /= scalar @$array1;
642 my $mean2 = sum(@{ $array2 });
643 $mean2 /= scalar @$array2;
644 return 1.0 if $mean1 == $mean2;
645 my ($variance1, $variance2) = (0, 0);
646 foreach my $x (@$array1) {
647 $variance1 += ($x-$mean1)*($x-$mean1);
648 }
649 foreach my $x (@$array2) {
650 $variance2 += ($x-$mean2)*($x-$mean2);
651 }
652 if (($variance1 == 0.0) && ($variance2 == 0.0)) {
653 return 1.0
654 }
655 $variance1 = $variance1/(scalar @$array1-1);
656 $variance2 = $variance2/(scalar @$array2-1);
657 my $array1_size = scalar @$array1;
658 my $array2_size = scalar @$array2;
659 my $WELCH_T_STATISTIC = ($mean1-$mean2)/sqrt($variance1/$array1_size+$variance2/$array2_size);
660 my $DEGREES_OF_FREEDOM = (($variance1/$array1_size+$variance2/(scalar @$array2))**2)
661 /
662 (
663 ($variance1*$variance1)/($array1_size*$array1_size*($array1_size-1))+
664 ($variance2*$variance2)/($array2_size*$array2_size*($array2_size-1))
665 );
666 my $A = $DEGREES_OF_FREEDOM/2;
667 my $value = $DEGREES_OF_FREEDOM/($WELCH_T_STATISTIC*$WELCH_T_STATISTIC+$DEGREES_OF_FREEDOM);
668#from here, translation of John Burkhardt's C
669 my $beta = lgamma($A)+0.57236494292470009-lgamma($A+0.5);
670 my $acu = 10**(-15);
671 my($ai,$cx,$indx,$ns,$pp,$psq,$qq,$rx,$temp,$term,$xx);
672# Check the input arguments.
673 return $value if $A <= 0.0;# || $q <= 0.0;
674 return $value if $value < 0.0 || 1.0 < $value;
675# Special cases
676 return $value if $value == 0.0 || $value == 1.0;
677 $psq = $A + 0.5;
678 $cx = 1.0 - $value;
679 if ($A < $psq * $value) {
680 ($xx, $cx, $pp, $qq, $indx) = ($cx, $value, 0.5, $A, 1);
681 } else {
682 ($xx, $pp, $qq, $indx) = ($value, $A, 0.5, 0);
683 }
684 $term = 1.0;
685 $ai = 1.0;
686 $value = 1.0;
687 $ns = int($qq + $cx * $psq);
688#Soper reduction formula.
689 $rx = $xx / $cx;
690 $temp = $qq - $ai;
691 $rx = $xx if $ns == 0;
692 while (1) {
693 #print "\$value = $value\n";
694 $term = $term * $temp * $rx / ( $pp + $ai );
695 #print "\$term = $term";
696 $value = $value + $term;
697 $temp = abs ($term);
698 if ($temp <= $acu && $temp <= $acu * $value) {
699# print "pp = $pp";
700# print "xx = $xx";
701# print "qq = $qq";
702# print "cx = $cx";
703# print "beta = $beta";
704 $value = $value * exp ($pp * log($xx)
705 + ($qq - 1.0) * log($cx) - $beta) / $pp;
706# print "the value = $value";
707 $value = 1.0 - $value if $indx;# == 0;
708# if ($indx == 0) {
709# $value = 1.0 - $value
710# }
711# print "\$indx = $indx";
712 last;
713 }
714 $ai = $ai + 1.0;
715 $ns = $ns - 1;
716 #print "ai = $ai";
717 #print "ns = $ns";
718 if (0 <= $ns) {
719 $temp = $qq - $ai;
720 $rx = $xx if $ns == 0;
721 } else {
722 $temp = $psq;
723 $psq = $psq + 1.0;
724 }
725 }
726 $value
727}
